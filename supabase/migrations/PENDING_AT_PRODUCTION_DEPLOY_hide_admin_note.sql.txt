-- DEFERRED — apply ONLY as part of the Production deploy of this branch.
--
-- Saved with a .txt extension and a PENDING_ prefix so `supabase db push`
-- skips it ("file name must match pattern <timestamp>_name.sql").
--
-- WHY IT IS DEFERRED
-- Customers can currently read admin_note on their own orders, because the
-- LIVE Production admin console (old website code) still selects it straight
-- from public.orders as the `authenticated` role — the same role customers
-- use. Column privileges are per role, so hiding admin_note from customers
-- today would also break the live admin order list.
--
-- The new website reads the admin order list through admin_list_orders()
-- (added in 20260918041253), which returns admin_note to admins only. Once
-- THAT website is live in Production, nothing needs table-level access to
-- admin_note any more, and this can be applied.
--
-- ORDER AT DEPLOY TIME
--   1. Deploy the new website to Production (it calls admin_list_orders()).
--   2. Confirm the Production admin order list loads.
--   3. Rename this file to <timestamp>_hide_admin_note_from_customers.sql
--      and `supabase db push`.
--   4. Confirm the admin order list still loads (now via the function only).
--
-- The shipped Expo app is unaffected either way: its admin screen never
-- selected admin_note.

revoke select on table public.orders from authenticated;

grant select (
  id,
  order_number,
  user_id,
  customer_email,
  customer_name,
  customer_phone,
  shipping_address,
  items,
  amount_total,
  currency,
  status,
  paid_at,
  created_at,
  updated_at,
  shipping_company,
  tracking_number,
  shipping_address_id,
  shipping_country_code,
  shipping_country_name,
  shipping_region,
  shipping_postal_code,
  product_amount,
  shipping_amount,
  shipping_city
) on table public.orders to authenticated;
